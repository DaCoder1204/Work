import java.util.GregorianCalendar; import java.util.Objects; public class Time { private int hours; private int minutes; private int seconds; public Time() { GregorianCalendar calendar = new GregorianCalendar(); this.hours = calendar.get(GregorianCalendar.HOUR_OF_DAY); this.minutes = calendar.get(GregorianCalendar.MINUTE); this.seconds = calendar.get(GregorianCalendar.SECOND); } public Time(int hours, int minutes, int seconds) { this.hours = hours; this.minutes = minutes; this.seconds = seconds; } public int getHours() { return hours; } public int getMinutes() { return minutes; } public int getSeconds() { return seconds; } @Override public String toString() { return hours + ":" + minutes + ":" + seconds; } public int incrementSeconds() { seconds++; fixMinutes(); return seconds; } private void fixMinutes() { if (seconds == 60) { seconds = 0; incrementMinutes(); } } public int incrementMinutes() { minutes++; fixHours(); return minutes; } private void fixHours() { if (minutes == 60) { minutes = 0; incrementHours(); } } public int incrementHours() { hours++; if (hours == 24) { hours = 0; } return hours; } @Override public boolean equals(Object o) { if (this == o) return true; if (o == null || getClass() != o.getClass()) return false; Time time = (Time) o; return hours == time.hours && minutes == time.minutes && seconds == time.seconds; } @Override public int hashCode() { return Objects.hash(hours, minutes, seconds); } }
