public class JacobList2<T> {

    private class Node {
        T data;
        Node next;
        Node rprevious;

        Node(T d) {
            data = d;
            next = null;
            rprevious = null;
        }
    }

    private Node head;
    private int count;

    public JacobList2() {
        head = null;
        count = 0;
    }

    public void add(T value) {
        Node n = new Node(value);

        if (head == null) {
            head = n;
        } else {
            Node temp = head;
            while (temp.next != null) {
                temp = temp.next;
            }

            temp.next = n;
            n.rprevious = temp;
        }

        count++;
    }

    // ---------------------------------------------------------
    // removeAt(index)
    // ---------------------------------------------------------
    public void removeAt(int index) {
        if (index < 0 || index >= count) {
            System.out.println("Invalid remove index: " + index);
            return;
        }

        // removing the head
        if (index == 0) {
            head = head.next;
            if (head != null) {
                head.rprevious = null;
            }
            count--;
            return;
        }

        Node temp = head;

        // move to the node BEFORE the one we want to remove
        for (int i = 0; i < index - 1; i++) {
            temp = temp.next;
        }

        Node toRemove = temp.next;

        // skip over the removed node
        temp.next = toRemove.next;

        if (toRemove.next != null) {
            toRemove.next.rprevious = temp;
        }

        count--;
    }

    // ---------------------------------------------------------
    // remove(value)
    // ---------------------------------------------------------
    public void remove(T value) {
        if (value == null) {
            System.out.println("Cannot remove null value.");
            return;
        }

        // removing head
        if (head != null && head.data.equals(value)) {
            head = head.next;
            if (head != null) {
                head.rprevious = null;
            }
            count--;
            return;
        }

        Node temp = head;

        // search for the node BEFORE the one we want to remove
        while (temp != null && temp.next != null) {
            if (temp.next.data.equals(value)) {
                Node toRemove = temp.next;

                temp.next = toRemove.next;

                if (toRemove.next != null) {
                    toRemove.next.rprevious = temp;
                }

                count--;
                return;
            }
            temp = temp.next;
        }

        System.out.println("Value not found: " + value);
    }

    public int size() {
        return count;
    }

    public T get(int index) {
        if (index < 0 || index >= count) {
            System.out.println("Invalid get index: " + index);
            return null;
        }

        Node temp = head;
        for (int i = 0; i < index; i++) {
            temp = temp.next;
        }

        return temp.data;
    }

    public void print() {
        Node temp = head;
        int i = 0;

        while (temp != null) {
            System.out.println("Index " + i + ": " + temp.data);
            temp = temp.next;
            i++;
        }
    }

    public void printReverse() {
        if (head == null) {
            System.out.println("List is empty");
            return;
        }

        Node temp = head;
        while (temp.next != null) {
            temp = temp.next;
        }

        int index = count - 1;
        while (temp != null) {
            System.out.println("Index " + index + ": " + temp.data);
            temp = temp.rprevious;
            index--;
        }
    }

    public void iterateForward() {
        boolean hasNext = (head != null);
        Node temp = head;

        while (hasNext) {
            System.out.println("Visiting: " + temp.data);

            temp = temp.next;

            if (temp == null) {
                hasNext = false;
            }
        }
    }
}
