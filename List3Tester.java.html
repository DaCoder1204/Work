public class List3Tester {
    public static void main(String[] args) {

        List3<String> list = new List3<>();

        System.out.println("=== Adding items ===");
        list.add("Ben");
        System.out.println("Size: " + list.size());

        list.add("James");
        System.out.println("Size: " + list.size());

        list.add("Charlie");
        System.out.println("Size: " + list.size());

        list.add("Harry");
        System.out.println("Size: " + list.size());

        list.add("Eric");
        System.out.println("Size: " + list.size());

        System.out.println("\n=== Current list (print) ===");
        list.print();

        System.out.println("\n=== Testing get ===");
        System.out.println("Item at index 0: " + list.get(0));
        System.out.println("Item at index 2: " + list.get(2));
        System.out.println("Item at index 4: " + list.get(4));

        System.out.println("\n=== Reverse print ===");
        list.printReverse();

        System.out.println("\n=== Forward iteration ===");
        list.iterateForward();

        // ---------------------------------------------------------
        // TAIL TESTING SECTION
        // ---------------------------------------------------------
        System.out.println("\n=== Testing tail behavior ===");

        System.out.println("Expected tail = Eric");
        System.out.println("Actual last element (get(size-1)) = " + list.get(list.size() - 1));

        System.out.println("\nTesting circular forward link (tail.next == head):");
        System.out.println("Head = " + list.get(0));
        System.out.println("After looping through list, last element = " + list.get(list.size() - 1));
        System.out.println("One more step should return to head = " + list.get(0));

        System.out.println("\nTesting circular backward link (head.rprevious == tail):");
        System.out.println("Reverse print:");
        list.printReverse();

        // ---------------------------------------------------------
        // REMOVE TESTS
        // ---------------------------------------------------------
        System.out.println("\n=== Removing by index (removeAt) ===");
        System.out.println("Removing index 2...");
        list.removeAt(2);
        System.out.println("Size: " + list.size());
        list.print();

        System.out.println("\nRemoving index 0...");
        list.removeAt(0);
        System.out.println("Size: " + list.size());
        list.print();

        System.out.println("\n=== Removing by value (remove(value)) ===");
        System.out.println("Removing 'Harry'...");
        list.remove("Harry");
        System.out.println("Size: " + list.size());
        list.print();

        System.out.println("\nTrying to remove 'NotHere'...");
        list.remove("NotHere");

        // ---------------------------------------------------------
        // FINAL TAIL CHECK AFTER REMOVALS
        // ---------------------------------------------------------
        System.out.println("\n=== Final tail verification ===");
        System.out.println("Expected tail = " + list.get(list.size() - 1));
        System.out.println("Reverse print (should end at new head):");
        list.printReverse();

        System.out.println("\n=== Final forward iteration ===");
        list.iterateForward();

        System.out.println("\nIf no infinite loops and prints look correct, tail is functioning properly.");
    }
}
