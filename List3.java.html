public class List3<T> {

    private class Node {
        T data;
        Node next;
        Node rprevious;

        Node(T d) {
            data = d;
            next = null;
            rprevious = null;
        }
    }

    private Node head;
    private Node tail;   // NEW: last node
    private int count;

    public List3() {
        head = null;
        tail = null;
        count = 0;
    }

    // ---------------------------------------------------------
    // ADD — maintains circular structure
    // ---------------------------------------------------------
    public void add(T value) {
        Node n = new Node(value);

        if (head == null) {
            head = n;
            tail = n;

            // circular links
            head.next = head;
            head.rprevious = head;
        } else {
            // attach after tail
            tail.next = n;
            n.rprevious = tail;

            // update tail
            tail = n;

            // close the circle
            tail.next = head;
            head.rprevious = tail;
        }

        count++;
    }

    // ---------------------------------------------------------
    // REMOVE AT INDEX
    // ---------------------------------------------------------
    public void removeAt(int index) {
        if (index < 0 || index >= count) {
            System.out.println("Invalid remove index: " + index);
            return;
        }

        // removing head
        if (index == 0) {
            if (count == 1) {
                head = null;
                tail = null;
                count = 0;
                return;
            }

            head = head.next;
            head.rprevious = tail;
            tail.next = head;

            count--;
            return;
        }

        Node temp = head;

        // move to node BEFORE the one to remove
        for (int i = 0; i < index - 1; i++) {
            temp = temp.next;
        }

        Node toRemove = temp.next;

        temp.next = toRemove.next;
        toRemove.next.rprevious = temp;

        // if removing tail, update tail
        if (toRemove == tail) {
            tail = temp;
        }

        count--;
    }

    // ---------------------------------------------------------
    // REMOVE BY VALUE
    // ---------------------------------------------------------
    public void remove(T value) {
        if (value == null) {
            System.out.println("Cannot remove null value.");
            return;
        }

        if (head == null) {
            System.out.println("List is empty.");
            return;
        }

        // removing head
        if (head.data.equals(value)) {
            removeAt(0);
            return;
        }

        Node temp = head;

        // loop until we return to head (circular)
        do {
            if (temp.next.data.equals(value)) {
                Node toRemove = temp.next;

                temp.next = toRemove.next;
                toRemove.next.rprevious = temp;

                if (toRemove == tail) {
                    tail = temp;
                }

                count--;
                return;
            }

            temp = temp.next;

        } while (temp != head);

        System.out.println("Value not found: " + value);
    }

    // ---------------------------------------------------------
    // SIZE
    // ---------------------------------------------------------
    public int size() {
        return count;
    }

    // ---------------------------------------------------------
    // GET
    // ---------------------------------------------------------
    public T get(int index) {
        if (index < 0 || index >= count) {
            System.out.println("Invalid get index: " + index);
            return null;
        }

        Node temp = head;

        for (int i = 0; i < index; i++) {
            temp = temp.next;
        }

        return temp.data;
    }

    // ---------------------------------------------------------
    // PRINT FORWARD (STOP AFTER count STEPS)
    // ---------------------------------------------------------
    public void print() {
        if (head == null) {
            System.out.println("List is empty");
            return;
        }

        Node temp = head;
        int i = 0;

        while (i < count) {
            System.out.println("Index " + i + ": " + temp.data);
            temp = temp.next;
            i++;
        }
    }

    // ---------------------------------------------------------
    // PRINT REVERSE (STOP AFTER count STEPS)
    // ---------------------------------------------------------
    public void printReverse() {
        if (tail == null) {
            System.out.println("List is empty");
            return;
        }

        Node temp = tail;
        int index = count - 1;

        while (index >= 0) {
            System.out.println("Index " + index + ": " + temp.data);
            temp = temp.rprevious;
            index--;
        }
    }

    // ---------------------------------------------------------
    // ITERATE FORWARD SAFELY
    // ---------------------------------------------------------
    public void iterateForward() {
        if (head == null) {
            System.out.println("List is empty");
            return;
        }

        Node temp = head;
        int steps = 0;

        while (steps < count) {
            System.out.println("Visiting: " + temp.data);
            temp = temp.next;
            steps++;
        }
    }
}
