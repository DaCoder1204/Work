public class List3 { private class Node { T data; Node next; Node previous; Node(T d) { data = d; next = null; previous = null; } } private Node head; private Node tail; private int count; public List3() { head = null; tail = null; count = 0; } public void add(T value) { Node n = new Node(value); if (head == null) { head = n; tail = n; // circular link head.next = head; head.previous = head; } else { // attaches after tail tail.next = n; n.previous = tail; // updates tail tail = n; // closes the circle tail.next = head; head.previous = tail; } count++; } // REMOVES AT THE INDEX public void removeAt(int index) { if (index < 0 || index >= count) { System.out.println("Invalid remove index: " + index); return; } // removes head if (index == 0) { if (count == 1) { head = null; tail = null; count = 0; return; } head = head.next; head.previous = tail; tail.next = head; count--; return; } Node temp = head; // moves temp to node before the one to remove for (int i = 0; i < index - 1; i++) { temp = temp.next; } Node toRemove = temp.next; temp.next = toRemove.next; toRemove.next.previous = temp; // if removing tail, updates the tail if (toRemove == tail) { tail = temp; } count--; } // REMOVES BY VALUE public void remove(T value) { if (value == null) { System.out.println("Cannot remove null value."); return; } if (head == null) { System.out.println("List is empty."); return; } // removes head if (head.data.equals(value)) { removeAt(0); return; } Node temp = head; // loop until return to head do { if (temp.next.data.equals(value)) { Node toRemove = temp.next; temp.next = toRemove.next; toRemove.next.previous = temp; if (toRemove == tail) { tail = temp; } count--; return; } temp = temp.next; } while (temp != head); System.out.println("Value not found: " + value); } public int size() { return count; } public T get(int index) { if (index < 0 || index >= count) { System.out.println("Invalid get index: " + index); return null; } Node temp = head; for (int i = 0; i < index; i++) { temp = temp.next; } return temp.data; } // PRINTS FORWARD public void print() { if (head == null) { System.out.println("List is empty"); return; } Node temp = head; int i = 0; while (i < count) { System.out.println("Index " + i + ": " + temp.data); temp = temp.next; i++; } } // PRINTS REVERSE public void printReverse() { if (tail == null) { System.out.println("List is empty"); return; } Node temp = tail; int index = count - 1; while (index >= 0) { System.out.println("Index " + index + ": " + temp.data); temp = temp.previous; index--; } } // ITERATES FORWARDS public void iterateForward() { if (head == null) { System.out.println("List is empty"); return; } Node temp = head; int steps = 0; while (steps < count) { System.out.println("Visiting: " + temp.data); temp = temp.next; steps++; } } }
