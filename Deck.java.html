public class Deck { private T[] data; private int size; public Deck() { data = (T[]) new Object [10]; size = 0; } public int size() { return size; } public void grow() { T[] bigger = (T[]) new Object[data.length * 2]; for (int i = 0; i < size; i++) { bigger[i] = data[i]; } data = bigger; } //adds element to top public void push(T value) { if (size == data.length) { grow(); } data[size] = value; size++; } //Removes element from top public T pop() { if (size == 0) { return null; } size--; T value = data[size]; data[size] = null; return value; } //ToString method for generics allows to print the deck @Override public String toString() { StringBuilder sb = new StringBuilder("["); for (int i = 0; i < size; i++) { sb.append(data[i]); if (i < size - 1) sb.append(", "); } sb.append("]"); return sb.toString(); } //finds number in the deck public boolean find(T tValue) { Deck temp = new Deck<>(); boolean found = false; // Pop everything to search while (this.size > 0) { T value = this.pop(); if (value.equals(tValue)) { found = true; } temp.push(value); } // Restores the original deck while (temp.size > 0) { this.push(temp.pop()); } return found; } }
