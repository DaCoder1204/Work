public class Deck <T> {
    private T[] data;
    private int size;

    public Deck() {
        data = (T[]) new Object [10];
        size = 0;
    }

    public int size() {
        return size;
    }

    public void grow() {
        T[] bigger = (T[]) new Object[data.length * 2];
        for (int i = 0; i < size; i++) {
            bigger[i] = data[i];
        }
        data = bigger;
    }

    //Push add element to top
    public void push(T value) {
        if (size == data.length) {
            grow();
        }
        data[size] = value;
        size++;
    }
    //Pop remove element from top
    public T pop() {
        if (size == 0) {
            return null;
        }
        size--;
        T value = data[size];
        data[size] = null;
        return value;
    }

    //ToString methods for generics to print deck
    @Override
    public String toString() {
        StringBuilder sb = new StringBuilder("[");
        for (int i = 0; i < size; i++) {
            sb.append(data[i]);
            if (i < size - 1) sb.append(", ");
        }
        sb.append("]");
        return sb.toString();
    }

    //Find method finds a given number in the deck
    public boolean find(T target) {
        Deck<T> temp = new Deck<>();

        boolean found = false;

        // Pop everything to search
        while (this.size > 0) {
            T value = this.pop();

            if (value.equals(target)) {
                found = true;
            }

            temp.push(value);
        }

        // Restore original deck
        while (temp.size > 0) {
            this.push(temp.pop());
        }

        return found;
    }
}
