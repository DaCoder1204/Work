public class DeckTester {
    public static void main(String [] args){
        Deck<Integer> deck = new Deck<>();

        System.out.println("\n====Start of Deck Test====");

        //Push values into array
        System.out.println("\n====Pushing in numbers 10, 20, 30====");
        deck.push(10);
        deck.push(20);
        deck.push(30);
        System.out.println(deck);

        // Test find
        System.out.println("\n====Testing find====");
        System.out.println("Find 20 (true): " + deck.find(20));
        System.out.println("Find 99 (false): " + deck.find(99));
        System.out.println("Deck after find: " + deck);

        //Pop one number
        System.out.println("\n====Popping top number====");
        Integer popped = deck.pop();
        System.out.println("Popped: " + popped);
        System.out.println(deck);

        //Push in more numbers
        System.out.println("\n====Pushing in 40 and 50====");
        deck.push(40);
        deck.push(50);
        System.out.println(deck);

        // Test find again
        System.out.println("\n====Testing find again====");
        System.out.println("Find 50 (true): " + deck.find(50));
        System.out.println("Find 10 (true): " + deck.find(10));
        System.out.println("Find 123 (false): " + deck.find(123));
        System.out.println("Deck after find: " + deck);

        //Pop everything in the array
        System.out.println("\n====Popping everything in array====");
        while (deck.size() > 0) {
            System.out.println("====Popped: " + deck.pop());
        }
        System.out.println(deck);

        System.out.println("\n==== Deck Test Complete ====");

        System.out.println("Current size: " + deck.size());
    }
}
